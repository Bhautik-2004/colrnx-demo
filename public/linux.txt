# ==============================================================
# TASK 1: N-GRAM MODELING
# ==============================================================
# Dataset:
#   Artificial Intelligence improves healthcare.
#   Artificial Intelligence is transforming education.
#   Machine learning improves healthcare.
#
# Tasks:
#   - Convert all text to lowercase.
#   - Tokenize the text.
#   - Generate: Unigrams, Bigrams, Trigrams
#   - Display the frequency of each N-gram.
#   - Calculate the Unigram probability for each word.
#   - Calculate the Bigram probability.
#   - Calculate the Trigram probability.
#   - Identify the most frequent: Unigram, Bigram, Trigram
# ==============================================================

from collections import Counter

# Dataset
text = """
Artificial Intelligence improves healthcare.
Artificial Intelligence is transforming education.
Machine learning improves healthcare.
"""

# -------------------------------
# Step 1: Lowercase
# -------------------------------
text = text.lower()

# -------------------------------
# Step 2: Tokenization
# -------------------------------
tokens = text.replace(".", "").split()

print("Tokens:")
print(tokens)

# -------------------------------
# Step 3: Generate Unigrams
# -------------------------------
unigrams = tokens

# Generate Bigrams
bigrams = [(tokens[i], tokens[i+1]) for i in range(len(tokens)-1)]

# Generate Trigrams
trigrams = [(tokens[i], tokens[i+1], tokens[i+2]) for i in range(len(tokens)-2)]

# -------------------------------
# Step 4: Frequency
# -------------------------------
uni_freq = Counter(unigrams)
bi_freq = Counter(bigrams)
tri_freq = Counter(trigrams)

print("\nUnigram Frequency")
for k,v in uni_freq.items():
    print(k,":",v)

print("\nBigram Frequency")
for k,v in bi_freq.items():
    print(k,":",v)

print("\nTrigram Frequency")
for k,v in tri_freq.items():
    print(k,":",v)

# -------------------------------
# Step 5: Unigram Probability
# -------------------------------
total_words = len(tokens)

print("\nUnigram Probability")
for word,count in uni_freq.items():
    print(word,"=",count/total_words)

# -------------------------------
# Step 6: Bigram Probability
# P(w2|w1)=Count(w1,w2)/Count(w1)
# -------------------------------
print("\nBigram Probability")

for bg,count in bi_freq.items():
    w1,w2 = bg
    prob = count/uni_freq[w1]
    print(bg,"=",prob)

# -------------------------------
# Step 7: Trigram Probability
# P(w3|w1,w2)=Count(w1,w2,w3)/Count(w1,w2)
# -------------------------------
print("\nTrigram Probability")

for tg,count in tri_freq.items():
    bg = (tg[0],tg[1])
    prob = count/bi_freq[bg]
    print(tg,"=",prob)

# -------------------------------
# Step 8: Most Frequent
# -------------------------------
print("\nMost Frequent Unigram:")
print(uni_freq.most_common(1))

print("\nMost Frequent Bigram:")
print(bi_freq.most_common(1))

print("\nMost Frequent Trigram:")
print(tri_freq.most_common(1))

"""

for token in doc:
    print(token.text)

# --------------------------------
# POS Tagging
# --------------------------------
print("\nPOS TAGGING")

verb_count = 0

    print(token.text, ":", token.pos_)

        verb_count += 1

# --------------------------------
# Named Entity Recognition
# --------------------------------
print("\nNAMED ENTITIES")

entity_count = 0

for ent in doc.ents:
    print(ent.text, ":", ent.label_)
    entity_count += 1

print("\nTotal Nouns =", noun_count)
print("Total Verbs =", verb_count)
print("Total Named Entities =", entity_count)

# --------------------------------
# PERSON
# --------------------------------
print("\nPERSON")
for ent in doc.ents:
    if ent.label_ == "PERSON":
        print(ent.text)

# --------------------------------
# ORG
# --------------------------------
print("\nORG")
for ent in doc.ents:
    if ent.label_ == "ORG":
        print(ent.text)

# --------------------------------
# GPE
# --------------------------------
print("\nGPE")
for ent in doc.ents:
    if ent.label_ == "GPE":
        print(ent.text)

# --------------------------------
# DATE
# --------------------------------
print("\nDATE")
for ent in doc.ents:

# ==============================================================
# TASK 1 — Q3: FULL NLP PIPELINE
# ==============================================================
# Sentence: "Plato's ideas are deep !!!Visit http://philosophy.com.
#            Philosophers discuss morality."
#
# Tasks:
#   a) Preprocessing pipeline:
#      - Lowercase, Remove HTML tags, Remove URLs,
#      - Remove punctuation, Tokenize, Remove stopwords,
#      - Stemming (Porter), Lemmatization (spaCy)
#   b) Bag of Words (BoW) and TF-IDF vectorization.
#   c) POS Tagging and Named Entity Recognition.
# ==============================================================

import re
import string
import spacy

from nltk.tokenize import word_tokenize
from nltk.corpus import stopwords
from nltk.stem import PorterStemmer

from sklearn.feature_extraction.text import CountVectorizer, TfidfVectorizer

# Load spaCy model
nlp = spacy.load("en_core_web_sm")

text = """Plato's ideas are deep !!!Visit http://philosophy.com.<div>Philosophers
discuss morality</div> ."""

# a) Preprocessing Pipeline
# Lowercase
processed = text.lower()

# Remove HTML tags
processed = re.sub(r'<.*?>', '', processed)

# Remove URLs
processed = re.sub(r'http\S+|www\S+', '', processed)

# Remove punctuation
processed = processed.translate(str.maketrans('', '', string.punctuation + "'"))

# Tokenize
tokens = word_tokenize(processed)

# Stopword removal
stop_words = set(stopwords.words('english'))
tokens = [word for word in tokens if word not in stop_words]

# Stemming
stemmer = PorterStemmer()
stemmed = [stemmer.stem(word) for word in tokens]

# Lemmatization
doc = nlp(" ".join(tokens))
lemmatized = [token.lemma_ for token in doc]

print("Tokens:", tokens)
print("Stemmed:", stemmed)
print("Lemmatized:", lemmatized)

# b) Bag of Words
corpus = [" ".join(lemmatized)]

cv = CountVectorizer()
bow = cv.fit_transform(corpus)

print("\nBoW Vocabulary:")
print(cv.get_feature_names_out())

print("BoW Matrix:")
print(bow.toarray())

# TF-IDF
tfidf = TfidfVectorizer()
tfidf_matrix = tfidf.fit_transform(corpus)

print("\nTF-IDF Vocabulary:")
print(tfidf.get_feature_names_out())

print("TF-IDF Matrix:")
print(tfidf_matrix.toarray())

# c) POS Tagging
doc = nlp(text)

print("\nPOS Tagging")
for token in doc:
    print(token.text, "->", token.pos_)

# Named Entity Recognition
print("\nNamed Entities")
if doc.ents:
    for ent in doc.ents:
        print(ent.text, "->", ent.label_)
else:
    print("No Named Entities Found.")

# TASK 3 — CODE 1: SENTIMENT ANALYSIS USING TEXTBLOB
# ==============================================================
# Dataset: 5 tweets
#   - I love this phone.
#   - This movie is boring.
#   - The service is okay.
#   - Amazing experience!
#   - Worst app ever.
#
# Task:
#   - Use TextBlob to compute polarity.
#   - Classify each tweet as Positive, Negative, or Neutral.
# ==============================================================

from textblob import TextBlob

# Dataset
tweets = [
    "I love this phone.",
    "This movie is boring.",
    "The service is okay.",
    "Amazing experience!",
    "Worst app ever."
]

# Sentiment Analysis
for tweet in tweets:

    blob = TextBlob(tweet)
    polarity = blob.sentiment.polarity

    if polarity > 0:
        sentiment = "Positive"
    elif polarity < 0:
        sentiment = "Negative"
    else:
        sentiment = "Neutral"

    print("Tweet      :", tweet)
    print("Polarity   :", polarity)
    print("Sentiment  :", sentiment)
    print("-" * 50)

# ==============================================================
# Dataset: 5 tweets
#   - Excellent service.
#   - The exam was horrible.

#   - I absolutely love ChatGPT!
#   - Very disappointing experience.
#
# Task:
#   - Use VADER SentimentIntensityAnalyzer.
#   - Print Negative, Neutral, Positive, and Compound scores.
#   - Classify as Positive (compound >= 0.05),
#     Negative (compound <= -0.05), or Neutral.
# ==============================================================

from vaderSentiment.vaderSentiment import SentimentIntensityAnalyzer

# Dataset
tweets = [
    "Excellent service.",
    "The exam was horrible.",
    "The weather is okay.",
    "I absolutely love ChatGPT!",
    "Very disappointing experience."
]

analyzer = SentimentIntensityAnalyzer()

for tweet in tweets:

    score = analyzer.polarity_scores(tweet)

    if score["compound"] >= 0.05:
        sentiment = "Positive"
    elif score["compound"] <= -0.05:
        sentiment = "Negative"
    else:
        sentiment = "Neutral"

    print("Tweet           :", tweet)
    print("Negative score  :", score["neg"])
    print("Neutral score   :", score["neu"])
    print("Positive score  :", score["pos"])
    print("Compound score  :", score["compound"])
    print("Sentiment       :", sentiment)
    print("-" * 60)

        textblob_sentiment = "Neutral"

    # -------------------------
    # VADER
    # -------------------------
    score = analyzer.polarity_scores(tweet)

    if score["compound"] >= 0.05:
        vader_sentiment = "Positive"
    elif score["compound"] <= -0.05:
        vader_sentiment = "Negative"
    else:
        vader_sentiment = "Neutral"

    # -------------------------
    # Print comparison
    # -------------------------
    print("=" * 60)
    print("Tweet :", tweet)

    print(
        "TextBlob :",
        textblob_sentiment,
        "| Polarity =",
        polarity
    )

    print(
        "VADER    :",
        vader_sentiment,
        "| Compound =",
        score["compound"]
    )
