def add_noise(images):
    noise = torch.randn_like(images) * 0.5
    noisy_images = images + noise

    return torch.clamp(noisy_images, 0, 1)

class Autoencoder(nn.Module):
    def __init__(self):
        super().__init__()

        self.encoder = nn.Sequential(
            nn.Flatten(),
            nn.Linear(28 * 28, 128),
            nn.ReLU(),
            nn.Linear(128, 64),
            nn.ReLU()
        )

        self.decoder = nn.Sequential(
            nn.Linear(64, 128),
            nn.ReLU(),
            nn.Linear(128, 28 * 28),
            nn.Sigmoid()
        )

    def forward(self, x):
        x = self.encoder(x)
        x = self.decoder(x)
        x = x.view(-1, 1, 28, 28)

        return x

# Variational Autoencoder for MNIST
# Simple PyTorch Implementation

import torch
import torch.nn as nn
import torch.optim as optim
from torchvision.datasets import MNIST
from torchvision import transforms
import matplotlib.pyplot as plt


# --------------------------------------------------
# 1. Device
# --------------------------------------------------

device = "cuda" if torch.cuda.is_available() else "cpu"
print("Device:", device)


# --------------------------------------------------
# 2. MNIST Dataset
# --------------------------------------------------

train_data = MNIST(
    "./data",
    train=True,
    download=True,
    transform=transforms.ToTensor()
)

train_loader = torch.utils.data.DataLoader(
    train_data,
    batch_size=128,
    shuffle=True
)


# --------------------------------------------------
# 3. VAE Model
# --------------------------------------------------

class VAE(nn.Module):

    def __init__(self):

        super().__init__()

        # Encoder
        self.encoder = nn.Sequential(
            nn.Flatten(),
            nn.Linear(784, 400),
            nn.ReLU(),
            nn.Linear(400, 200),
            nn.ReLU()
        )

        # Latent space
        self.fc_mu = nn.Linear(200, 20)
        self.fc_logvar = nn.Linear(200, 20)

        # Decoder
        self.decoder = nn.Sequential(
            nn.Linear(20, 200),
            nn.ReLU(),
            nn.Linear(200, 400),
            nn.ReLU(),
            nn.Linear(400, 784),
            nn.Sigmoid()
        )


    # Reparameterization trick
    def reparameterize(self, mu, logvar):

        std = torch.exp(0.5 * logvar)

        epsilon = torch.randn_like(std)

        z = mu + epsilon * std

        return z


    def forward(self, x):

        # Encoder
        x = self.encoder(x)

        # Mean and log variance
        mu = self.fc_mu(x)
        logvar = self.fc_logvar(x)

        # Sample latent vector
        z = self.reparameterize(mu, logvar)

        # Decoder
        output = self.decoder(z)

        return output, mu, logvar


# --------------------------------------------------
# 4. Create Model
# --------------------------------------------------

model = VAE().to(device)

optimizer = optim.Adam(
    model.parameters(),
    lr=0.001
)


# --------------------------------------------------
# 5. VAE Loss
# --------------------------------------------------

def vae_loss(output, images, mu, logvar, beta):

    images = images.view(-1, 784)

    # Reconstruction loss
    reconstruction_loss = nn.functional.binary_cross_entropy(
        output,
        images,
        reduction="sum"
    )

    # KL Divergence
    kl_loss = -0.5 * torch.sum(
        1 + logvar - mu.pow(2) - logvar.exp()
    )

    # Total loss
    loss = reconstruction_loss + beta * kl_loss

    # Average over batch
    return loss / images.size(0)


# --------------------------------------------------
# 6. Training
# --------------------------------------------------

epochs = 10

for epoch in range(epochs):

    model.train()

    total_loss = 0

    # Gradually increase KL weight
    # Helps the VAE learn good reconstructions first
    # and then learn a proper latent distribution.

    beta = min(1.0, (epoch + 1) / 10)

    for images, _ in train_loader:

        images = images.to(device)

        # Forward pass
        output, mu, logvar = model(images)

        # Loss
        loss = vae_loss(
            output,
            images,
            mu,
            logvar,
            beta
        )

        # Backpropagation
        optimizer.zero_grad()

        loss.backward()

        optimizer.step()

        total_loss += loss.item()

    avg_loss = total_loss / len(train_loader)

    print(
        f"Epoch [{epoch + 1}/{epochs}] "
        f"Loss: {avg_loss:.2f} "
        f"Beta: {beta:.2f}"
    )


# --------------------------------------------------
# 7. Generate New Digits
# --------------------------------------------------

model.eval()

with torch.no_grad():

    # Random samples from standard normal distribution
    z = torch.randn(10, 20).to(device)

    # Decode
    generated_images = model.decoder(z)

    # Reshape to 28 x 28
    generated_images = generated_images.view(
        -1, 1, 28, 28
    )

    generated_images = generated_images.cpu()


# --------------------------------------------------
# 8. Display Generated Images
# --------------------------------------------------

plt.figure(figsize=(8, 4))

for i in range(10):

    plt.subplot(2, 5, i + 1)

    plt.imshow(
        generated_images[i].squeeze(),
        cmap="gray"
    )

    plt.axis("off")
    plt.title("Generated")

plt.tight_layout()
plt.show()

import torch
import torch.nn as nn
import torch.optim as optim
from torchvision import datasets, transforms
from torch.utils.data import DataLoader

# --------------------------------------------------
# 1. Device
# --------------------------------------------------

device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
print("Device:", device)


# --------------------------------------------------
# 2. Fashion MNIST
# --------------------------------------------------

transform = transforms.ToTensor()

train_data = datasets.FashionMNIST(
    "./data",
    train=True,
    download=True,
    transform=transform
)

test_data = datasets.FashionMNIST(
    "./data",
    train=False,
    download=True,
    transform=transform
)

train_loader = DataLoader(
    train_data,
    batch_size=128,
    shuffle=True
)

test_loader = DataLoader(
    test_data,
    batch_size=128,
    shuffle=False
)


# --------------------------------------------------
# 3. CNN Model
# --------------------------------------------------

class CNN(nn.Module):

    def __init__(self, dropout=False):
        super().__init__()

        self.features = nn.Sequential(

            # Layer 1
            nn.Conv2d(1, 32, kernel_size=3, padding=1),
            nn.ReLU(),
            nn.MaxPool2d(2),

            # Layer 2
            nn.Conv2d(32, 64, kernel_size=3, padding=1),
            nn.ReLU(),
            nn.MaxPool2d(2),

            # Layer 3
            nn.Conv2d(64, 128, kernel_size=3, padding=1),
            nn.ReLU(),

            # Layer 4
            nn.Conv2d(128, 256, kernel_size=3, padding=1),
            nn.ReLU(),

            nn.MaxPool2d(2)
        )

        self.dropout = nn.Dropout(0.5) if dropout else nn.Identity()

        self.classifier = nn.Sequential(
            nn.Flatten(),
            nn.Linear(256 * 3 * 3, 128),
            nn.ReLU(),
            nn.Linear(128, 10)
        )

    def forward(self, x):
        x = self.features(x)
        x = self.dropout(x)
        x = self.classifier(x)
        return x


# --------------------------------------------------
# 4. Train + Test function
# --------------------------------------------------

def train_model(model, l1=False, l2=False, epochs=5):

    model = model.to(device)

    loss_fn = nn.CrossEntropyLoss()

    # L2 is implemented using weight_decay
    if l2:
        optimizer = optim.Adam(
            model.parameters(),
            lr=0.001,
            weight_decay=0.0001
        )
    else:
        optimizer = optim.Adam(
            model.parameters(),
            lr=0.001
        )

    for epoch in range(epochs):

        model.train()

        correct = 0
        total = 0

        for images, labels in train_loader:

            images = images.to(device)
            labels = labels.to(device)

            output = model(images)

            loss = loss_fn(output, labels)

            # L1 regularization
            if l1:
                l1_loss = 0

                for parameter in model.parameters():
                    l1_loss += torch.sum(torch.abs(parameter))

                loss = loss + 0.00001 * l1_loss

            optimizer.zero_grad()

            loss.backward()

            optimizer.step()

            # Training accuracy
            predictions = output.argmax(dim=1)

            total += labels.size(0)
            correct += (predictions == labels).sum().item()

        train_accuracy = 100 * correct / total

        print(
            f"Epoch {epoch + 1}/{epochs} "
            f"- Train Accuracy: {train_accuracy:.2f}%"
        )

    # --------------------------------------------------
    # Test
    # --------------------------------------------------

    model.eval()

    correct = 0
    total = 0

    with torch.no_grad():

        for images, labels in test_loader:

            images = images.to(device)
            labels = labels.to(device)

            output = model(images)

            predictions = output.argmax(dim=1)

            total += labels.size(0)
            correct += (predictions == labels).sum().item()

    test_accuracy = 100 * correct / total

    return train_accuracy, test_accuracy


# --------------------------------------------------
# 5. Run all 5 models
# --------------------------------------------------

results = {}


# A. Base Model
print("\n==============================")
print("A. BASE MODEL")
print("==============================")

Q11. VAE — Generate Handwritten Digit Images
──────────────────────────────────────────────────────────────────────

import tensorflow as tf
from tensorflow.keras import layers, Model
import numpy as np
import matplotlib.pyplot as plt

# ── 1. Load and Prepare MNIST ──────────────────────────────
(x_train, _), (x_test, _) = tf.keras.datasets.mnist.load_data()

x_train = x_train.astype("float32") / 255.
x_test  = x_test.astype("float32")  / 255.

x_train = x_train.reshape(-1, 784)
x_test  = x_test.reshape(-1, 784)

input_dim  = 784
latent_dim = 16

# ── 2. Encoder ─────────────────────────────────────────────
encoder_inputs = layers.Input(shape=(input_dim,), name="Encoder_Input")

x = layers.Dense(512, activation="relu")(encoder_inputs)
x = layers.Dense(256, activation="relu")(x)
x = layers.Dense(128, activation="relu")(x)

z_mean    = layers.Dense(latent_dim, name="Latent_Mean")(x)
z_log_var = layers.Dense(latent_dim, name="Latent_LogVar")(x)


# ── 3. Reparameterization Trick ────────────────────────────
def sampling(args):
    z_mean, z_log_var = args
    epsilon = tf.random.normal(shape=tf.shape(z_mean))
    return z_mean + tf.exp(0.5 * z_log_var) * epsilon


z = layers.Lambda(sampling, name="Sampling")([z_mean, z_log_var])

encoder = Model(encoder_inputs, [z_mean, z_log_var, z], name="Encoder")
encoder.summary()


# ── 4. Decoder ─────────────────────────────────────────────
latent_inputs = layers.Input(shape=(latent_dim,), name="Decoder_Input")

x = layers.Dense(128, activation="relu")(latent_inputs)
x = layers.Dense(256, activation="relu")(x)
x = layers.Dense(512, activation="relu")(x)

decoder_outputs = layers.Dense(input_dim, activation="sigmoid")(x)

decoder = Model(latent_inputs, decoder_outputs, name="Decoder")
decoder.summary()


# ── 5. VAE Model ───────────────────────────────────────────
class VAE(Model):
    def __init__(self, encoder, decoder):
        super(VAE, self).__init__()
        self.encoder = encoder
        self.decoder = decoder

    def call(self, inputs):
        z_mean, z_log_var, z = self.encoder(inputs)
        reconstructed = self.decoder(z)

        # Reconstruction Loss
        reconstruction_loss = tf.reduce_mean(
            tf.keras.losses.binary_crossentropy(inputs, reconstructed)) * 784

        # KL Divergence Loss
        kl_loss = -0.5 * tf.reduce_mean(
            tf.reduce_sum(
                1 + z_log_var - tf.square(z_mean) - tf.exp(z_log_var),
                axis=1
            )
        )

        self.add_loss(reconstruction_loss + kl_loss)
        return reconstructed


vae = VAE(encoder, decoder)


# ── 6. Compile ─────────────────────────────────────────────
vae.compile(optimizer="adam")


# ── 7. Train ───────────────────────────────────────────────
vae.fit(
    x_train, x_train,
    epochs=30,
    batch_size=64,
    validation_data=(x_test, x_test)
)


# ── 8. Reconstruction Example ──────────────────────────────
reconstructed = vae.predict(x_test[:10])

plt.figure(figsize=(12, 4))
for i in range(10):
    plt.subplot(2, 10, i+1)
    plt.imshow(x_test[i].reshape(28, 28), cmap="gray")
    plt.axis("off")

    plt.subplot(2, 10, i+11)
    plt.imshow(reconstructed[i].reshape(28, 28), cmap="gray")
    plt.axis("off")

plt.show()

import tensorflow as tf

x_test  = x_test.reshape(-1, 28, 28, 1).astype("float32")  / 255.0


# 3. Function to create CNN model with different regularization
def create_model(model_type):

    regularizer = None
    use_dropout = False

    if model_type == "L1":
        regularizer = l1(0.001)
    elif model_type == "L2":
        regularizer = l2(0.001)
    elif model_type == "Dropout":
        use_dropout = True
    elif model_type == "L2+Dropout":
        regularizer = l2(0.001)
        use_dropout = True

    model = Sequential()

    # 4 Convolution Layers
    model.add(Conv2D(32, (3, 3), activation='relu', padding='same',
                     kernel_regularizer=regularizer, input_shape=(28, 28, 1)))
    model.add(Conv2D(32, (3, 3), activation='relu', padding='same',
                     kernel_regularizer=regularizer))
    model.add(MaxPooling2D((2, 2)))

    model.add(Conv2D(64, (3, 3), activation='relu', padding='same',
                     kernel_regularizer=regularizer))
    model.add(Conv2D(64, (3, 3), activation='relu', padding='same',
                     kernel_regularizer=regularizer))
    model.add(MaxPooling2D((2, 2)))

    model.add(Flatten())
    model.add(Dense(128, activation='relu', kernel_regularizer=regularizer))

    if use_dropout:
        model.add(Dropout(0.5))

    model.add(Dense(10, activation='softmax'))

    model.compile(
        optimizer='adam',
        loss='sparse_categorical_crossentropy',
        metrics=['accuracy']
    )
    return model

                        validation_split=0.2, verbose=1)

    train_acc = history.history["accuracy"][-1]

import tensorflow as tf
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Conv2D, MaxPooling2D, Flatten, Dense, Dropout
import time
import matplotlib.pyplot as plt


# 1. Create CNN Model
def create_model(input_shape):
    model = Sequential([
        Conv2D(32, (3, 3), activation='relu', padding='same', input_shape=input_shape),
        Conv2D(32, (3, 3), activation='relu', padding='same'),
        MaxPooling2D((2, 2)),

        Conv2D(64, (3, 3), activation='relu', padding='same'),
        Conv2D(64, (3, 3), activation='relu', padding='same'),
        MaxPooling2D((2, 2)),

        Flatten(),
        Dense(128, activation='relu'),
        Dropout(0.5),
        Dense(10, activation='softmax')
    ])

    model.compile(
        optimizer='adam',
        loss='sparse_categorical_crossentropy',
        metrics=['accuracy']
    )
    return model


# 2. Experiment function
def run_experiment(dataset_name):

    print("\n" + "="*60)
    print("Dataset:", dataset_name)
    print("="*60)

    # Load dataset
    if dataset_name == "MNIST":
        (x_train, y_train), (x_test, y_test) = tf.keras.datasets.mnist.load_data()
    elif dataset_name == "Fashion MNIST":
        (x_train, y_train), (x_test, y_test) = tf.keras.datasets.fashion_mnist.load_data()

    # Preprocess
    x_train = x_train.reshape(-1, 28, 28, 1).astype("float32") / 255.0

    results = []

    for epochs in epochs_list:

        print(f"\nTraining for {epochs} epochs")

        model = create_model((28, 28, 1))

        start = time.time()
        model.fit(x_train, y_train, epochs=epochs, batch_size=64,
                  validation_split=0.2, verbose=1)
        end   = time.time()

        loss, accuracy = model.evaluate(x_test, y_test, verbose=0)
        runtime = end - start

        results.append([epochs, accuracy * 100, runtime])

        print(f"Accuracy : {accuracy*100:.2f}%")
        print(f"Time     : {runtime:.2f} seconds")

    # Print result table
    print("\nFinal Results")
    print("---------------------------------------------")
    print("Epochs\tAccuracy (%)\tTime (sec)")
    print("---------------------------------------------")
    for r in results:
        print(f"{r[0]}\t{r[1]:.2f}\t\t{r[2]:.2f}")

    # Plot
    epoch_values = [r[0] for r in results]
    accuracies   = [r[1] for r in results]

    plt.xlabel("Epochs")
    plt.ylabel("Accuracy (%)")
    plt.grid(True)
    plt.show()
