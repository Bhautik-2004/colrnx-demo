# ==============================================================
# TASK 1 — Q1: SINGLE NEURON WITH SIGMOID ACTIVATION
# ==============================================================
# Inputs: [2, 3] | Weights: [0.5, -0.4] | Bias: 0.2
#
# Task:
#   - Compute weighted sum z = sum(x*w) + b.
#   - Apply sigmoid activation and print neuron output.
# ==============================================================

import math

def sigmoid(x):
    return 1 / (1 + math.exp(-x))

def single_neuron(inputs, weights, bias):
    z = 0
    for i in range(len(inputs)):
        z += inputs[i] * weights[i]
    z += bias
    return sigmoid(z)

inputs = [2, 3]
weights = [0.5, -0.4]
bias = 0.2

result = single_neuron(inputs, weights, bias)
print("Neuron Output:", result)

# TASK 1 — Q2: PERCEPTRON FOR AND / OR LOGIC GATES
# ==============================================================
# Task:
#   - Implement perceptron with step activation.
#   - Use weights/bias to realise AND gate and OR gate.
#   - Print output for all 4 input combinations.
# ==============================================================

def perceptron(x, w, b):
    return 1 if np.dot(x, w) + b >= 0 else 0

OR_weights = np.array([1, 1])
OR_bias = -0.5

inputs = [[0, 0], [0, 1], [1, 0], [1, 1]]

print("AND Gate")
for x in inputs:
    print(x, perceptron(x, AND_weights, AND_bias))

print("\nOR Gate")

# ==============================================================
# TASK 1 — Q11: VARIATIONAL AUTOENCODER (VAE) FOR DIGIT GENERATION
# ==============================================================
# Dataset: MNIST
#
# Task:
#   - Build encoder (z_mean, z_log_var) with reparameterization sampling.
#   - Build decoder and custom VAE train_step (Reconstruction + KL loss).
#   - Generate new digits from a 5x5 latent space grid.
# ==============================================================

import numpy as np
import matplotlib.pyplot as plt
import tensorflow as tf
from tensorflow.keras.models import Model
from tensorflow.keras.layers import Input, Dense, Lambda
from tensorflow.keras.datasets import mnist
import keras

print("--- Task 11: Variational Autoencoder (VAE) for Digit Generation ---")

# 1. Load and process data
(X_train, _), (_, _) = mnist.load_data()
X_images = X_train.reshape(-1, 784) / 255.0

latent_dim = 2

# 2. Stochastic sampling function structure
def sampling(args):
    z_mean, z_log_var = args
    batch = tf.shape(z_mean)[0]
    dim = tf.shape(z_mean)[1]
    epsilon = tf.random.normal(shape=(batch, dim))
    return z_mean + tf.exp(0.5 * z_log_var) * epsilon

# 3. Build Encoder
vae_input = Input(shape=(784,), name="encoder_input")
h = Dense(128, activation='relu')(vae_input)
z_mean = Dense(latent_dim, name="z_mean")(h)
z_log_var = Dense(latent_dim, name="z_log_var")(h)
z = Lambda(sampling, name="z")([z_mean, z_log_var])
encoder = Model(vae_input, [z_mean, z_log_var, z], name="encoder")

# 4. Build Decoder
latent_inputs = Input(shape=(latent_dim,), name="z_sampling")
h_decoded = Dense(128, activation='relu')(latent_inputs)
decoder_output = Dense(784, activation='sigmoid')(h_decoded)
decoder = Model(latent_inputs, decoder_output, name="decoder")

# 5. Build Keras 3 Compliant Custom Model Overriding train_step
class VAE(Model):
    def __init__(self, encoder, decoder, **kwargs):
        super().__init__(**kwargs)
        self.encoder = encoder
        self.decoder = decoder
        # Track metric averages smoothly over epochs
        self.total_loss_tracker = keras.metrics.Mean(name="total_loss")
        self.reconstruction_loss_tracker = keras.metrics.Mean(name="reconstruction_loss")
        self.kl_loss_tracker = keras.metrics.Mean(name="kl_loss")

    @property
    def metrics(self):
        return [self.total_loss_tracker, self.reconstruction_loss_tracker, self.kl_loss_tracker]

    def train_step(self, data):
        with tf.GradientTape() as tape:
            # Forward pass through encoder and decoder
            z_mean, z_log_var, z = self.encoder(data)
            reconstruction = self.decoder(z)
            
            # Use keras.losses inside your model context to remain framework safe
            recon_loss = tf.reduce_mean(
                tf.reduce_sum(keras.losses.binary_crossentropy(data, reconstruction), axis=-1)
            )
            kl_loss = -0.5 * tf.reduce_mean(
                tf.reduce_sum(1 + z_log_var - tf.square(z_mean) - tf.exp(z_log_var), axis=-1)
            )
            total_loss = recon_loss + kl_loss
            
        # Backward pass gradients
        grads = tape.gradient(total_loss, self.trainable_weights)
        self.optimizer.apply_gradients(zip(grads, self.trainable_weights))
        
        # Update metrics states
        self.total_loss_tracker.update_state(total_loss)
        self.reconstruction_loss_tracker.update_state(recon_loss)
        self.kl_loss_tracker.update_state(kl_loss)
        
        return {m.name: m.result() for m in self.metrics}

# 6. Instantiate and Train the custom model
vae = VAE(encoder, decoder)
vae.compile(optimizer='adam')
vae.fit(X_images, epochs=5, batch_size=128, verbose=1)

# 7. Generate and display new arbitrary digits directly via the latent grid
plt.figure(figsize=(6, 6))
grid_x = np.linspace(-1.5, 1.5, 5)
grid_y = np.linspace(-1.5, 1.5, 5)

count = 1
for i, yi in enumerate(grid_x):
    for j, xi in enumerate(grid_y):
        z_sample = np.array([[xi, yi]])
        generated_digit = decoder.predict(z_sample, verbose=0).reshape(28, 28)
        
        plt.subplot(5, 5, count)
        plt.imshow(generated_digit, cmap='gray')
        plt.axis('off')
        count += 1
plt.suptitle("Generated Digit Images from Random Latent Space Coordinates")
plt.show()

# ==============================================================
# TASK 2 — Q1: DEEP FEED FORWARD NEURAL NETWORK ON MNIST
# ==============================================================
# Dataset: MNIST
#
# Task:
#   - Build Deep FFNN (512 -> 256 -> 128 -> 10).
#   - Train for 5 and 10 epochs.
#   - Print train/test accuracy table and plot accuracy vs epochs.
# ==============================================================

import tensorflow as tf
from tensorflow.keras.datasets import mnist
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Dense, Flatten
import matplotlib.pyplot as plt

# Load dataset
(x_train, y_train), (x_test, y_test) = mnist.load_data()

# Normalize
x_train = x_train / 255.0
x_test = x_test / 255.0

# Build Deep Feed Forward Neural Network
model = Sequential([
    Flatten(input_shape=(28,28)),
    Dense(512, activation='relu'),
    Dense(256, activation='relu'),
    Dense(128, activation='relu'),
    Dense(10, activation='softmax')
])

model.compile(
    optimizer='adam',
    loss='sparse_categorical_crossentropy',
    metrics=['accuracy']
)

epochs_list = [5, 10]

train_acc = []
test_acc = []

for ep in epochs_list:
    print(f"\nTraining for {ep} epochs")
    history = model.fit(
        x_train,
        y_train,
        epochs=ep,
        batch_size=64,
        verbose=1
    )

    loss, acc = model.evaluate(x_test, y_test, verbose=0)

    train_acc.append(history.history['accuracy'][-1])
    test_acc.append(acc)

print("\nEpoch\tTrain Accuracy\tTest Accuracy")
for i in range(len(epochs_list)):
    print(f"{epochs_list[i]}\t{train_acc[i]:.4f}\t\t{test_acc[i]:.4f}")

plt.plot(epochs_list, test_acc, marker='o')
plt.title("MNIST Accuracy vs Epochs")
plt.xlabel("Epochs")

plt.grid(True)
plt.show()

    model = Sequential([

        x_train,

        epochs=ep,

    )

# CNN Model (4 Convolution Layers)
def create_cnn_model():

    model = Sequential([

        Conv2D(32, (3,3), activation='relu', padding='same', input_shape=(28,28,1)),
        Conv2D(32, (3,3), activation='relu', padding='same'),
        MaxPooling2D((2,2)),

        Conv2D(64, (3,3), activation='relu', padding='same'),
        Conv2D(64, (3,3), activation='relu', padding='same'),
        MaxPooling2D((2,2)),

        Flatten(),

        Dense(128, activation='relu'),
        Dropout(0.3),

        Dense(10, activation='softmax')
    ])

    model.compile(
        optimizer='adam',
        loss='sparse_categorical_crossentropy',
        metrics=['accuracy']
    )

    return model


epochs_list = [5, 10, 20]

datasets = [
    ("MNIST", mnist.load_data()),
    ("Fashion MNIST", fashion_mnist.load_data())
]


for dataset_name, data in datasets:

    print("\n=====================================")
    print("Dataset :", dataset_name)
    print("=====================================")

    (x_train, y_train), (x_test, y_test) = data

    # Normalize
    x_train = x_train.astype("float32") / 255.0
    x_test = x_test.astype("float32") / 255.0

    # Reshape
    x_train = x_train.reshape(-1, 28, 28, 1)
    x_test = x_test.reshape(-1, 28, 28, 1)

    accuracy_values = []
    time_values = []

    for ep in epochs_list:

        print("\nTraining for", ep, "epochs")

        model = create_cnn_model()

        start = time.time()

        model.fit(
            x_train,
            y_train,
            epochs=ep,
            batch_size=64,
            validation_split=0.1,
            verbose=1

# TASK 2 — Q5: REGULARIZATION COMPARISON (BASE / L1 / L2 / DROPOUT / L2 + DROPOUT)
# ==============================================================
# Dataset: Fashion-MNIST

# Task:
#   - Build 4-conv CNN with selected regularization.
#   - Train each model for 10 epochs.

import tensorflow as tf
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Conv2D, MaxPooling2D
from tensorflow.keras.layers import Flatten, Dense, Dropout
from tensorflow.keras.datasets import fashion_mnist
from tensorflow.keras.regularizers import l1, l2
import pandas as pd

# Load Fashion-MNIST
(x_train, y_train), (x_test, y_test) = fashion_mnist.load_data()

# Normalize
x_train = x_train.astype("float32") / 255.0
x_test = x_test.astype("float32") / 255.0

# Reshape
x_train = x_train.reshape(-1,28,28,1)
x_test = x_test.reshape(-1,28,28,1)


def create_model(model_type):

    if model_type == "Base":
        reg = None
        drop = 0

    elif model_type == "L1":
        reg = l1(0.001)
        drop = 0

    elif model_type == "L2":
        reg = l2(0.001)
        drop = 0

    elif model_type == "Dropout":
        reg = None
        drop = 0.5

    elif model_type == "L2 + Dropout":
        reg = l2(0.001)
        drop = 0.5

    model = Sequential()

    # 4 Convolution Layers
    model.add(Conv2D(32,(3,3),padding='same',
                     activation='relu',
                     kernel_regularizer=reg,
                     input_shape=(28,28,1)))

    model.add(Conv2D(32,(3,3),padding='same',
                     activation='relu',
                     kernel_regularizer=reg))

    model.add(MaxPooling2D((2,2)))

    model.add(Conv2D(64,(3,3),padding='same',
                     activation='relu',
                     kernel_regularizer=reg))

    model.add(Conv2D(64,(3,3),padding='same',
                     activation='relu',
                     kernel_regularizer=reg))

    model.add(MaxPooling2D((2,2)))

    model.add(Flatten())

    model.add(Dense(128,
                    activation='relu',
                    kernel_regularizer=reg))

    if drop > 0:
        model.add(Dropout(drop))

    model.add(Dense(10, activation='softmax'))

    model.compile(
        optimizer='adam',
        loss='sparse_categorical_crossentropy',
        metrics=['accuracy']
    )

    return model


models = [
    "Base",
    "L1",
    "L2",
    "Dropout",
    "L2 + Dropout"
]

results = []

for m in models:

    print("\n===========================")
    print("Training:", m)
    print("===========================")

    model = create_model(m)

    history = model.fit(
        x_train,
        y_train,
        epochs=10,
        batch_size=64,
        validation_split=0.1,
        verbose=1
    )

    loss, test_acc = model.evaluate(
        x_test,
        y_test,
        verbose=0
    )

    results.append([
        m,
        round(train_acc*100,2),
        round(test_acc*100,2)
    ])

)

# Parameters

num_classes = 10
m = 16                    # Number of filters
k = 3                     # Filter size
n = 128                   # Dense layer neurons
activation_function = 'relu'

# Build CNN
model = Sequential()

for i in range(5):
    if i == 0:
        model.add(Conv2D(m, (k, k), input_shape=input_shape, padding='same'))
    else:
        model.add(Conv2D(m, (k, k), padding='same'))
    model.add(Activation(activation_function))
    model.add(MaxPooling2D(pool_size=(2, 2)))

model.add(Flatten())
model.add(Dense(n))
model.add(Activation(activation_function))
model.add(Dense(num_classes))
model.add(Activation('softmax'))

# Display model
model.summary()

# --------------------------------------------------
# Calculations
# --------------------------------------------------

# After 5 max-pooling layers:
# 224 -> 112 -> 56 -> 28 -> 14 -> 7
flatten_size = 7 * 7 * m

# Computations (MACs)
conv1_computations = 224 * 224 * 3 * k * k * m
conv2_computations = 112 * 112 * m * k * k * m
conv3_computations = 56 * 56 * m * k * k * m
conv4_computations = 28 * 28 * m * k * k * m
conv5_computations = 14 * 14 * m * k * k * m

dense_computations = flatten_size * n
output_computations = n * num_classes

total_computations = (
    conv1_computations +
    conv2_computations +
    conv3_computations +
    conv4_computations +
    conv5_computations +
    dense_computations +
    output_computations
)

# Parameters
conv1_parameters = (k * k * 3 * m) + m
conv2_parameters = (k * k * m * m) + m
conv3_parameters = (k * k * m * m) + m
conv4_parameters = (k * k * m * m) + m
conv5_parameters = (k * k * m * m) + m

dense_parameters = (flatten_size * n) + n
output_parameters = (n * num_classes) + num_classes

total_parameters = (
    conv1_parameters +
    conv2_parameters +
    conv3_parameters +
    conv4_parameters +
    conv5_parameters +
    dense_parameters +
    output_parameters
)

# Print results
print("\n--- CNN Calculations ---")

print("Filter Size (k x k) :", k, "x", k)
print("Dense Neurons (n) :", n)
